CREATE TABLE CLIENTE(ID_CLIENTE INT AUTO_INCREMENT PRIMARY KEY, NOME VARCHAR(60), EMAIL VARCHAR(60), TELEFONE VARCHAR(30), SENHA VARCHAR(50));

CREATE TABLE ENDERECO(ID_ENDERECO INT AUTO_INCREMENT PRIMARY KEY, RUA VARCHAR(60), NUMERO VARCHAR(10), COMPLEMENTO VARCHAR(60), CEP VARCHAR(10), BAIRRO VARCHAR(30), CIDADE VARCHAR(30), ID_CLIENTE INT, FOREIGN KEY (ID_CLIENTE) REFERENCES CLIENTE(ID_CLIENTE));

CREATE TABLE PRODUTOS(ID_PRODUTO INT AUTO_INCREMENT PRIMARY KEY, NOME VARCHAR(64), TIPO VARCHAR(32), VALOR FLOAT, DESCRICAO VARCHAR(300), FOTO VARCHAR(300));

CREATE TABLE PIZZAS(ID_PIZZA INT AUTO_INCREMENT PRIMARY KEY, SABOR VARCHAR(64), TIPO VARCHAR(32), VALOR FLOAT, DESCRICAO VARCHAR(300), FOTO VARCHAR(300), DISPONIBILIDADE BOOLEAN);

CREATE TABLE PIZZAIOLO(ID_PIZZAIOLO INT AUTO_INCREMENT PRIMARY KEY, NOME VARCHAR(64), SENHA VARCHAR(64), EMAIL VARCHAR(64));

CREATE TABLE PAGAMENTO(ID_PAGAMENTO INT AUTO_INCREMENT PRIMARY KEY, FORMAPAGAMENTO VARCHAR(32));

CREATE TABLE itens_pedido (
    id_itenspedido  BIGINT PRIMARY KEY AUTO_INCREMENT,
    id_pedido       BIGINT NOT NULL,
    id_pizza        BIGINT NULL,
    id_produto      BIGINT NULL,
    quantidade      INT NOT NULL CHECK (quantidade > 0),
    preco_unitario  DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (id_pedido)  REFERENCES pedido(id_pedido),
    FOREIGN KEY (id_pizza)   REFERENCES pizza(id_pizza),
    FOREIGN KEY (id_produto) REFERENCES produtos(id_produto),
    -- exatamente um dos dois preenchido
    CHECK ((id_pizza IS NOT NULL AND id_produto IS NULL)
        OR (id_pizza IS NULL AND id_produto IS NOT NULL))
);

CREATE TABLE CARDAPIO(ID_CARDAPIO INT AUTO_INCREMENT PRIMARY KEY, ID_PRODUTO INT, ID_PIZZA INT,FOREIGN KEY(ID_PRODUTO) REFERENCES PRODUTOS(ID_PRODUTO), FOREIGN KEY (ID_PIZZA) REFERENCES PIZZAS(ID_PIZZA));

CREATE TABLE pedido (
    id_pedido     BIGINT PRIMARY KEY AUTO_INCREMENT,
    id_cliente    BIGINT NOT NULL,
    id_endereco   BIGINT NOT NULL,
    id_pagamento  BIGINT NOT NULL,
    horario       TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status        VARCHAR(20) NOT NULL DEFAULT 'RECEBIDO',
    valor_total   DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (id_cliente)   REFERENCES cliente(id_cliente),
    FOREIGN KEY (id_endereco)  REFERENCES endereco(id_endereco),
    FOREIGN KEY (id_pagamento) REFERENCES pagamento(id_pagamento)
);
