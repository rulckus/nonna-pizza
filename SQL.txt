CREATE TABLE CLIENTE(ID_CLIENTE INT AUTO_INCREMENT PRIMARY KEY, NOME VARCHAR(60), EMAIL VARCHAR(60), TELEFONE VARCHAR(30), SENHA VARCHAR(50));

CREATE TABLE ENDERECO(ID_ENDERECO INT AUTO_INCREMENT PRIMARY KEY, RUA VARCHAR(60), NUMERO VARCHAR(10), COMPLEMENTO VARCHAR(60), CEP VARCHAR(10), BAIRRO VARCHAR(30), CIDADE VARCHAR(30), ID_CLIENTE INT, FOREIGN KEY (ID_CLIENTE) REFERENCES CLIENTE(ID_CLIENTE));

CREATE TABLE PRODUTOS(ID_PRODUTO INT AUTO_INCREMENT PRIMARY KEY, NOME VARCHAR(64), TIPO VARCHAR(32), VALOR FLOAT, DESCRICAO VARCHAR(300), FOTO VARCHAR(300));

CREATE TABLE TAMANHOS(ID_TAMANHO INT AUTO_INCREMENT PRIMARY KEY, TAMANHOS VARCHAR(60));

CREATE TABLE PIZZAS(ID_PIZZA INT AUTO_INCREMENT PRIMARY KEY, SABOR VARCHAR(64), TIPO VARCHAR(32), VALOR FLOAT, DESCRICAO VARCHAR(300), ID_TAMANHO INT, FOTO VARCHAR(300), DIPONIBILIDADE BOOLEAN, FOREIGN KEY (ID_TAMANHO) REFERENCES TAMANHO(ID_TAMANHO));

CREATE TABLE PIZZAIOLO(ID_PIZZAIOLO INT AUTO_INCREMENT PRIMARY KEY, NOME VARCHAR(64), SENHA VARCHAR(64), EMAIL VARCHAR(64));

CREATE TABLE PAGAMENTO(ID_PAGAMENTO INT AUTO_INCREMENT PRIMARY KEY, FORMAPAGAMENTO VARCHAR(32));

CREATE TABLE ITENSPEDIDO(ID_ITENSPEDIDO INT AUTO_INCREMENT PRIMARY KEY, ID_PRODUTO INT, ID_PIZZA INT, FOREIGN KEY(ID_PRODUTO) REFERENCES PRODUTOS(ID_PRODUTO), FOREIGN KEY (ID_PIZZA) REFERENCES PIZZAS(ID_PIZZA));

CREATE TABLE CARDAPIO(ID_CARDAPIO INT AUTO_INCREMENT PRIMARY KEY, ID_PRODUTO INT, ID_PIZZA INT,FOREIGN KEY(ID_PRODUTO) REFERENCES PRODUTOS(ID_PRODUTO), FOREIGN KEY (ID_PIZZA) REFERENCES PIZZAS(ID_PIZZA));

CREATE TABLE PEDIDO(ID_PEDIDO INT AUTO_INCREMENT PRIMARY KEY, ID_PAGAMENTO INT, ID_ITENSPEDIDO INT, HORARIO DATETIME, STATUS VARCHAR(64), ID_CLIENTE INT, FOREIGN KEY (ID_PAGAMENTO) REFERENCES PAGAMENTO(ID_PAGAMENTO), FOREIGN KEY (ID_ITENSPEDIDO) REFERENCES ITENSPEDIDO(ID_ITENSPEDIDO), FOREIGN KEY (ID_CLIENTE) REFERENCES CLIENTE(ID_CLIENTE));
